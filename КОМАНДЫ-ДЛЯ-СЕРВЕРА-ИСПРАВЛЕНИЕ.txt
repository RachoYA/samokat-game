=================================================================
🔧 ИСПРАВЛЕНИЕ ДЕПЛОЯ - УСТАНОВКА PM2
=================================================================

ПРОБЛЕМА: PM2 не установлен на сервере

РЕШЕНИЕ: Выполните эти команды на сервере


ШАГ 1: Установите PM2
-----------------------------------------------------------------

sudo npm install -g pm2


ШАГ 2: Запустите приложение
-----------------------------------------------------------------

cd /var/www/samokat-game

# Запуск Backend через PM2
pm2 start deployment/ecosystem.config.js

# Сохранить конфигурацию PM2
pm2 save

# Автозапуск PM2 при перезагрузке
pm2 startup
# Выполните команду, которую выведет pm2 startup (начинается с sudo)


ШАГ 3: Настройте Nginx
-----------------------------------------------------------------

# Установите Nginx (если не установлен)
sudo apt install -y nginx

# Скопируйте конфигурацию
sudo cp /var/www/samokat-game/deployment/nginx.conf /etc/nginx/sites-available/samokat-game

# Создайте симлинк
sudo ln -s /etc/nginx/sites-available/samokat-game /etc/nginx/sites-enabled/samokat-game

# Проверьте конфигурацию
sudo nginx -t

# Перезапустите Nginx
sudo systemctl reload nginx


ШАГ 4: Проверьте статус
-----------------------------------------------------------------

# Статус PM2
pm2 status
pm2 logs

# Статус Nginx
sudo systemctl status nginx


ШАГ 5: Откройте firewall (если нужно)
-----------------------------------------------------------------

sudo ufw allow 80/tcp
sudo ufw allow 443/tcp
sudo ufw status
